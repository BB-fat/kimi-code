342	    signal: AbortSignal,
343	  ): AsyncIterable<TimedToolResult> {

346	    const pendingResults = new Map<number, Promise<SettledTimedToolResult>>();
347	
348	    for (let index = 0; index < tasks.length; index += 1) {

597	function preflightToolCall(
598	  toolRegistry: IAgentToolRegistryService,
599	  toolCall: ToolCall,
600	  log?: ILogService,
601	): PreflightedToolCall {
602	  const toolName = toolCall.name;
603	  const parsedArgs = parseToolCallArguments(toolCall.arguments);
604	  if (parsedArgs.parseFailed) {
605	    log?.debug('tool args JSON parse failed', {
606	      toolName,
607	      toolCallId: toolCall.id,
608	      rawLength: typeof toolCall.arguments === 'string' ? toolCall.arguments.length : 0,
609	      error: parsedArgs.error,
610	    });
611	  }
612	  const tool = toolRegistry.resolve(toolName);
613	  if (tool === undefined) {
614	    return {
615	      kind: 'rejected',
616	      toolCall,
617	      toolName,
618	      args: parsedArgs.data,
619	      output: `Tool "${toolName}" not found`,
620	    };
621	  }
622	  const validationError = validateExecutableToolArgs(tool, parsedArgs.data);
623	  if (validationError !== null) {
624	    return {
625	      kind: 'rejected',
626	      toolCall,
627	      toolName,
628	      args: parsedArgs.data,
629	      output: `Invalid args for tool "${toolName}": ${validationError}`,
630	    };
631	  }
632	  return { kind: 'runnable', toolCall, toolName, tool, args: parsedArgs.data };
633	}

637	  readonly parseFailed: boolean;

639	} {
640	  if (raw === null || raw === undefined || (typeof raw === 'string' && raw.length === 0)) {
